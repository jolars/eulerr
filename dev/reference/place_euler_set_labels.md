``` r
  shape,
  h,
  k,
  a,

  phi,
  width,
  height,
  side,
  container_h,
  container_k,
  container_width,
  container_height,
  n_vertices,
  label_widths,
  label_heights,
  obstacle_h,
  obstacle_k,
  obstacle_width,
  obstacle_height,
  margin,
  angular_steps,
  precision
)
```
